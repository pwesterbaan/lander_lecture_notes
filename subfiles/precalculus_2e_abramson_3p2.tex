  \section{3.2: Quadratic functions}

  \begin{defn*}
    A quadratic is a function of degree two. The \textbf{general form of a quadratic function} is
      \[f(x)=ax^2+bx+c \quad (a\neq 0)\]
    where $a$, $b$, and $c$ represent constants. The \textbf{standard form of a quadratic} is 
      \[f(x)=a\parens{x-h}^2+k.\]
    A \textbf{parabola} is the shape of the graph of a quadratic function, with a vertex at $(h,k)$:
      \[h=-\dfrac{b}{2a}, \quad \textnormal{ and }\quad k=f\parens{h}=c-\dfrac{b^2}{4a}\]

    \TabPositions{50mm}
    The optimal value occurs at the vertex of a parabola:
    \begin{itemize}
      \item A maximum if $a<0$
        \tab\tikz{\begin{axis}[axis lines=none,height=20mm]
            \addplot[<->, lander_blue] expression[domain=-1:1]{-x^2};
          \end{axis}}
      \item A minimum if $a>0$
        \tab\tikz{\begin{axis}[axis lines=none,height=20mm]
            \addplot[<->, lander_blue] expression[domain=-1:1]{x^2};
          \end{axis}}
    \end{itemize}
  \end{defn*}

  \begin{ex*}
    Find the vertex of the following quadratic functions by
    \begin{itemize}
      \item computing $h$ and $k$, and
      \item rewriting the quadratic in standard form by completing the square.
    \end{itemize}
    Identify if the vertex is the maximum or minimum value.
    \begin{extasks}[after-item-skip=\stretch{1}](1)
      \task $f(x)=2x^2-6x+7$
      \task $g(x)=13-3x^2+4x$
    \end{extasks}
    \vspace*{\stretch{1}}
  \end{ex*}
