  \begin{thmBox*}
    The general forms of the other trigonometric functions are
    \begin{align*}
      y&=A\tan\parens{Bx-C}+D,&
      y&=A\cot\parens{Bx-C}+D,\\
      y&=A\sec\parens{Bx-C}+D,&
      y&=A\csc\parens{Bx-C}+D.
    \end{align*}
    where
    \begin{itemize}
      \item $A$ determines the scaling factor,
      \item $B$ determines the period,
      \item $C$ determines the phase shift, and
      \item $D$ determines the vertical shift.
    \end{itemize}
  \end{thmBox*}
  \begin{ex*}
    Draw one period of the function $y=\frac{1}{2}\tan\parens{\frac{\pi}{2}x}$
    \vspace*{\stretch{1}}

    \begin{flushright}
      \begin{tikzpicture}[
        declare function={
          g(\x)=tan(deg(\x*pi));
          d=0.4;
          },
          nodeLabel/.style={pos=0.5, inner sep=5pt},
          every pin/.style={text=black, fill=white, opacity=0.85, text opacity=1},
          every pin edge/.style={black!50, line width=0.5pt},
        ]
        \begin{axis}[
          group style={group size=2 by 2,horizontal sep=25mm,vertical sep=22.5mm},
          axis lines=center,
          axis line style={->},
          xmin=-0.55, xmax=0.55,
%          ymin=-2.5, ymax=4.5,
          xtick={-1,-0.5,...,1},
          xticklabels={$-\pi$, $-\frac{\pi}{2}$,  , $\frac{\pi}{2}$, $\pi$},
          minor x tick num=1, xlabel=,
          minor y tick num=1, ylabel=,
          enlargelimits={abs=0.15},
          height=0.5\textheight, width=\axisdefaultwidth,
          ticklabel style={font=\normalsize,inner sep=5pt,fill=white,opacity=1.0, text opacity=1},
          every axis plot/.append style={domain=-d:d},
          ]
          \addplot[<->] {g(x)};
          \draw[black!50, dashed] (axis cs: -0.5,{tan(deg(-d*pi))}) -- (-0.5,{tan(deg(d*pi))});
          \draw[black!50, dashed] (axis cs: 0.5,{tan(deg(-d*pi))}) -- (0.5,{tan(deg(d*pi))});
        \end{axis}
      \end{tikzpicture}
    \end{flushright}
  \end{ex*}
