  \section{1.6: Absolute Value Functions}

  \begin{defn*}[Absolute Value Function]
    The absolute value function can be defined as a piecewise function
      \[f(x)=\abs{x}=
      \begin{cases}
        \phantom{-}x, &\textnormal{ if } x\geq 0\\
        -x, &\textnormal{ if } x<0
      \end{cases}\]
  \end{defn*}
