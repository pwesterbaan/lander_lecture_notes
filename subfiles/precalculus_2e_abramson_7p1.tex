  \section{7.1: Simplifying and Verifying Trigonometric Identities}
  Recall the following trigonometric identities:
  \begin{thmBox*}
    \begin{itemize}
      \item Pythagorean Identities:
        \begin{align*}
          \cos^2(\theta)+\mathrlap{\sin^2}\phantom{\tan^2}(\theta)&=1\\[5pt]
          1+\tan^2(\theta)&=\sec^2(\theta)\\[5pt]
          \cot^2(\theta)+\mathrlap{1}\phantom{\tan^2(\theta)}&=\csc^2(\theta)
        \end{align*}
      \item Even-odd Identities
        \begin{center}
          \begin{tabular}{@{}c>{\hspace*{20mm}}c@{}}\toprule
            Even& Odd\\\midrule
            $\cos(-\theta)=\cos(\theta)$ & $\sin(-\theta)=-\sin(\theta)$\\[5pt]
            $\sec(-\theta)=\sec(\theta)$ & $\csc(-\theta)=-\csc(\theta)$\\[5pt]
              & $\tan(-\theta)=-\tan(\theta)$\\[5pt]
              & $\cot(-\theta)=-\cot(\theta)$\\
            \bottomrule
          \end{tabular}
        \end{center}
      \item Reciprocal identities:
        \begin{align*}
          \sin\parens{\theta}&= \dfrac{1}{\csc\parens{\theta}}& 
          \cos\parens{\theta}&= \dfrac{1}{\sec\parens{\theta}}& 
          \tan\parens{\theta}&= \dfrac{1}{\cot\parens{\theta}}\\[5pt]
          \csc\parens{\theta}&= \dfrac{1}{\sin\parens{\theta}}& 
          \sec\parens{\theta}&= \dfrac{1}{\cos\parens{\theta}}& 
          \cot\parens{\theta}&= \dfrac{1}{\tan\parens{\theta}}
        \end{align*}
      \item Quotient identities:
        \begin{align*}
          \tan\parens{\theta}&= \dfrac{\sin\parens{\theta}}{\cos\parens{\theta}}& 
          \cot\parens{\theta}&= \dfrac{\cos\parens{\theta}}{\sin\parens{\theta}}
        \end{align*}
    \end{itemize}
  \end{thmBox*}
  \pagebreak

  \begin{ex*}
    Verify $\tan(\theta)\cos(\theta)=\sin(\theta)$. \vspace*{\stretch{1}}

    Verify $\csc(\theta)\cos(\theta)\tan(\theta)=1$. \vspace*{\stretch{1}}
    
    Verify $\parens{1+\sin(x)}\parens{1+\sin(-x)}=\cos^2(x)$ \vspace*{\stretch{1}}
    
    Verify $\dfrac{\sec^2(\theta)-1}{\sec^2(\theta)}=\sin^2(\theta)$ \vspace*{\stretch{1}}
  \end{ex*}
