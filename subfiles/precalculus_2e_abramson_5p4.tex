  \section{5.4: Right Angle Trigonometry}
  \begin{defn*}
    Given a right triangle with an acute angle $\theta$
    \begin{center}
      \phantom{opposite}
      \begin{tikzpicture}[
        declare function={d=25; r=6.5;},
        line cap=round, line join=round,
        clip=false,
        ]
        \coordinate (A) at (0,0);
        \coordinate (B) at (d:r);
        \coordinate (C) at ({r*cos(d)},0);
        \draw (A) -- node[pos=0.65, above left] {hypotenuse}
              (B) -- node[pos=0.5, right] {opposite}
              (C) -- node[pos=0.5, below] {adjacent}
              cycle;

        \tkzMarkRightAngle[size=0.4](B,C,A);

        \draw[black,-] (0.2*r,0) arc [start angle=0, end angle=d, radius=0.2*r];
        \node at (d/2:0.25*r) {$\theta$};
      \end{tikzpicture}
    \end{center}
    we have
    \begin{align*}
      \sin(\theta)&=\dfrac{\textnormal{opposite}}{\textnormal{hypotenuse}}\\[10pt]
      \cos(\theta)&=\dfrac{\textnormal{adjacent}}{\textnormal{hypotenuse}}\\[10pt]
      \tan(\theta)&=\dfrac{\textnormal{opposite}}{\textnormal{adjacent}}
    \end{align*}
  \end{defn*}

  %cofunction of complements
  \textbf{Equal cofunctions of complements:}
  \begin{center}
    \begin{tikzpicture}[
      declare function={d=30; r=5.5;},
      line cap=round, line join=round,
      ]
      \coordinate (A) at (0,0);
      \coordinate (B) at (d:r);
      \coordinate (C) at ({r*cos(d)},0);
      \draw (A) -- (B) -- (C) -- cycle;

      \tkzMarkRightAngle[size=0.4](B,C,A);

      \draw[black,-] (0.2*r,0) arc [start angle=0, delta angle=d, radius=0.2*r];
      \node at (d/2:0.25*r) {$\alpha$};

      \draw[black,-] ($(B)+(-90:0.15*r)$) arc [start angle=-90, delta angle=-90+d, radius=0.15*r];
      \node at ($(B)+({-135+d/2}:0.225*r)$) {$\beta$};
    \end{tikzpicture}
  \end{center}
  \begin{align*}
    \sin(\alpha)&=\cos(\beta)\\
    \sin(\beta)&=\cos(\alpha)
  \end{align*}

  \begin{defn*}[Cofunction Identities]
    \begin{align*}
      \cos\parens{\theta}&=\sin\parens{\dfrac{\pi}{2}-\theta}&
      \sin\parens{\theta}&=\cos\parens{\dfrac{\pi}{2}-\theta}\\[10pt]
      \sec\parens{\theta}&=\csc\parens{\dfrac{\pi}{2}-\theta}&
      \csc\parens{\theta}&=\sec\parens{\dfrac{\pi}{2}-\theta}\\[10pt]
      \cot\parens{\theta}&=\tan\parens{\dfrac{\pi}{2}-\theta}&
      \tan\parens{\theta}&=\cot\parens{\dfrac{\pi}{2}-\theta}
    \end{align*}
  \end{defn*}
  \begin{ex*}
    If $\sin(t)=\dfrac{5}{12}$, find $\cos\parens{\dfrac{\pi}{2}-t}$.
    \vspace*{\stretch{1}}
  \end{ex*}
  \begin{ex*}
    If $\csc\parens{\dfrac{\pi}{6}}=2$, find $\sec\parens{\dfrac{\pi}{3}}$.
    \vspace*{\stretch{1}}
  \end{ex*}
